统一码（Unicode）字符串比较与序列比对。

Go 批量 APIgo run bible.go
package main

import (
    "bufio"
    "errors"
    "fmt"
    "io"
    "log"
    "net/http"
    "os"
    "strings"

    stridealign "github.com/adamdeprince/stride-align/bindings/go"
)

func main() {
    const corpus = "kjv.txt"
    if _, err := os.Stat(corpus); errors.Is(err, os.ErrNotExist) {
        response, err := http.Get(
            "https://openbible.com/textfiles/kjv.txt",
        )
        if err != nil {
            log.Fatal(err)
        }
        defer response.Body.Close()
        file, err := os.Create(corpus)
        if err != nil {
            log.Fatal(err)
        }
        if _, err = io.Copy(file, response.Body); err != nil {
            log.Fatal(err)
        }
        if err = file.Close(); err != nil {
            log.Fatal(err)
        }
    } else if err != nil {
        log.Fatal(err)
    }

    file, err := os.Open(corpus)
    if err != nil {
        log.Fatal(err)
    }
    defer file.Close()
    var references, verses []string
    scanner := bufio.NewScanner(file)
    for scanner.Scan() {
        line := strings.TrimPrefix(scanner.Text(), "\ufeff")
        reference, verse, ok := strings.Cut(line, "\t")
        if ok {
            references = append(references, reference)
            verses = append(verses, verse)
        }
    }
    if err := scanner.Err(); err != nil {
        log.Fatal(err)
    }

    normalized := make([]string, len(verses))
    for index, verse := range verses {
        normalized[index] = strings.ToLower(verse)
    }
    query := strings.Join([]string{
        "The earth will mourn, and the heavens will lament from above.",
        "For I have spoken, I have decided, and I have not regretted.",
        "Neither will I be turned away from it.",
    }, " ")
    scores, err := stridealign.NeedlemanWunschNormalizedScores(
        strings.ToLower(query), normalized, nil,
    )
    if err != nil || len(scores) == 0 {
        log.Fatal("could not score corpus: ", err)
    }
    best := 0
    for index := 1; index < len(scores); index++ {
        if scores[index] > scores[best] {
            best = index
        }
    }
    fmt.Printf(
        "%s\t%s\nscore: %.6f\n",
        references[best], verses[best], scores[best],
    )
}

import os
from pathlib import Path
from urllib.request import urlretrieve

import duckdb

corpus = Path("kjv.txt")
if not corpus.exists():
    urlretrieve("https://openbible.com/textfiles/kjv.txt", corpus)

db = duckdb.connect(config={"allow_unsigned_extensions": "true"})
extension = os.environ["STRIDE_ALIGN_DUCKDB_EXTENSION"].replace("'", "''")
db.execute(f"LOAD '{extension}'")

db.execute(
    """
    CREATE OR REPLACE TABLE demo_bible AS
    SELECT trim(reference) AS reference, trim(verse) AS verse
    FROM read_csv(
        ?,
        delim = '\t',
        header = false,
        skip = 2,
        columns = {'reference': 'VARCHAR', 'verse': 'VARCHAR'},
        quote = '',
        escape = ''
    )
    WHERE reference IS NOT NULL
      AND verse IS NOT NULL
      AND trim(verse) != ''
    """,
    [str(corpus.resolve())],
)

query = (
    "The earth will mourn, and the heavens will lament from above. "
    "For I have spoken, I have decided, and I have not regretted. "
    "Neither will I be turned away from it."
)

reference, verse, score = db.execute(
    """
    WITH corpus AS (
        SELECT
            list(reference ORDER BY reference, verse) AS reference_values,
            list(verse ORDER BY reference, verse) AS verse_values,
            list(lower(verse) ORDER BY reference, verse) AS normalized_verses
        FROM demo_bible
    ), matched AS (
        SELECT
            reference_values,
            verse_values,
            stride_extract_best(
                lower(?),
                normalized_verses,
                'needleman_wunsch_normalized'
            ) AS best_match
        FROM corpus
    )
    SELECT
        reference_values[cast(best_match.index AS BIGINT) + 1] AS reference,
        verse_values[cast(best_match.index AS BIGINT) + 1] AS verse,
        best_match.score AS score
    FROM matched
    """,
    [query],
).fetchone()

print(f"{reference}\t{verse}")
print(f"score: {score:.6f}")

Python 编排方式与 Python 示例一样：已有文件就复用，否则下载；随后将语料库载入 DuckDB。SQL 形成一个有序候选列表，再由 stride_extract_best 执行 stride-align 优化的 Needleman–Wunsch 批量与最佳一项路径。

import os
from pathlib import Path
from urllib.request import urlretrieve

import psycopg

corpus = Path("kjv.txt")
if not corpus.exists():
    urlretrieve("https://openbible.com/textfiles/kjv.txt", corpus)

rows = [
    (reference.strip(), verse.strip())
    for line in corpus.read_text(encoding="utf-8-sig").splitlines()
    if "\t" in line
    for reference, verse in [line.split("\t", 1)]
    if verse.strip()
]
query = (
    "The earth will mourn, and the heavens will lament from above. "
    "For I have spoken, I have decided, and I have not regretted. "
    "Neither will I be turned away from it."
)

with psycopg.connect(os.environ["STRIDE_ALIGN_POSTGRES_DSN"]) as db:
    db.execute(
        "CREATE TEMP TABLE demo_bible "
        "(reference text, verse text) ON COMMIT DROP"
    )
    with db.cursor().copy(
        "COPY demo_bible (reference, verse) FROM STDIN"
    ) as copy:
        for row in rows:
            copy.write_row(row)

    reference, verse, score = db.execute(
        """
        WITH corpus AS (
            SELECT
                array_agg(reference ORDER BY reference, verse) AS references,
                array_agg(verse ORDER BY reference, verse) AS verses,
                array_agg(lower(verse) ORDER BY reference, verse) AS normalized
            FROM demo_bible
        ), matched AS (
            SELECT references, verses, stride_extract_best(
                lower(%s), normalized, 'needleman_wunsch_normalized'
            ) AS best_match
            FROM corpus
        )
        SELECT
            references[((best_match->>'index')::integer) + 1],
            verses[((best_match->>'index')::integer) + 1],
            (best_match->>'score')::double precision
        FROM matched
        """,
        (query,),
    ).fetchone()

print(f"{reference}\t{verse}")
print(f"score: {score:.6f}")

Python 通过 PostgreSQL 的 COPY 协议装载语料库。有序的 text[] 只进入 stride_extract_best 一次，使 stride-align 优化过的批量与最佳一项路径留在服务器内执行。

把同一操作用于拼写检查器。

这个示例有意保持简单，不处理标点、词频、上下文或专有名词。它只用于展示如何通过原生批量路径或数据库记录流，把一个词元与整部词典比较。

Python 批量 APIprintf '...' | python3 spell.py
import sys
from pathlib import Path

import stride_align as sa

paths = [
    Path("/usr/share/dict/words"),
    Path("/usr/dict/words"),
    Path("/var/lib/dict/words"),
    Path("/etc/dictionaries-common/words"),
]
dictionary_path = next(path for path in paths if path.is_file())
words = list(dict.fromkeys(
    word.strip().lower()
    for word in dictionary_path.read_text(errors="ignore").splitlines()
    if word.strip()
))

def correct(token):
    scores = sa.needleman_wunsch_normalized_scores(token, words)
    return words[int(scores.argmax())]

tokens = sys.stdin.readline().lower().split()
print(" ".join(correct(token) for token in tokens))

每个词元都会对整部词典发起一次批量调用；评分循环留在原生库中。

library(stridealign)

paths <- c(
    "/usr/share/dict/words",
    "/usr/dict/words",
    "/var/lib/dict/words",
    "/etc/dictionaries-common/words"
)
dictionary_path <- paths[file.exists(paths)][[1L]]

words <- words[nzchar(words)]

correct <- function(token) {
    scores <- needleman_wunsch_normalized_scores(token, words)
    words[[which.max(scores)]]
}

line <- readLines("stdin", n = 1L, warn = FALSE)
tokens <- strsplit(tolower(line), "\\s+")[[1L]]
cat(paste(vapply(tokens, correct, ""), collapse = " "), "\n")

package main

import (
    "bufio"
    "fmt"
    "log"
    "os"
    "strings"

    stridealign "github.com/adamdeprince/stride-align/bindings/go"
)

func main() {
    paths := []string{
        "/usr/share/dict/words",
        "/usr/dict/words",
        "/var/lib/dict/words",
        "/etc/dictionaries-common/words",
    }
    var dictionary *os.File
    for _, path := range paths {
        file, err := os.Open(path)
        if err == nil {
            dictionary = file
            break
        }
    }
    if dictionary == nil {
        log.Fatal("no system dictionary found")
    }
    defer dictionary.Close()

    seen := make(map[string]struct{})
    var words []string
    scanner := bufio.NewScanner(dictionary)
    for scanner.Scan() {
        word := strings.ToLower(strings.TrimSpace(scanner.Text()))
        if _, exists := seen[word]; word != "" && !exists {
            seen[word] = struct{}{}
            words = append(words, word)
        }
    }
    if err := scanner.Err(); err != nil {
        log.Fatal(err)
    }

    input := bufio.NewScanner(os.Stdin)
    if !input.Scan() {
        return
    }
    tokens := strings.Fields(strings.ToLower(input.Text()))
    corrected := make([]string, len(tokens))
    for index, token := range tokens {
        match, ok, err := stridealign.Best(
            token,
            words,
            stridealign.ScorerNeedlemanWunschNormalized,
            nil,
        )
        if err != nil || !ok {
            log.Fatal("could not match token: ", err)
        }
        corrected[index] = match.Target
    }
    fmt.Println(strings.Join(corrected, " "))
}

每个词元都会把完整词典切片交给 stride-align 的有界原生最佳一项路径；Go 无需构造或排序完整分数向量。

import os
import sys
from pathlib import Path

from gqlalchemy import Memgraph

paths = [
    Path("/usr/share/dict/words"),
    Path("/usr/dict/words"),
    Path("/var/lib/dict/words"),
    Path("/etc/dictionaries-common/words"),
]
dictionary_path = next((path for path in paths if path.is_file()), None)
if dictionary_path is None:
    raise FileNotFoundError("no system dictionary found")

words = sorted({
    word.strip().lower()
    for word in dictionary_path.read_text(errors="ignore").splitlines()
    if word.strip()
})
tokens = sys.stdin.readline().lower().split()

db = Memgraph(
    host=os.environ.get("STRIDE_ALIGN_MEMGRAPH_HOST", "127.0.0.1"),
    port=int(os.environ.get("STRIDE_ALIGN_MEMGRAPH_PORT", "7687")),
)
db.execute("MATCH (w:StrideAlignDemoWord) DETACH DELETE w")
for offset in range(0, len(words), 1000):
    db.execute(
        """UNWIND $words AS word
        CREATE (:StrideAlignDemoWord {value: word})""",
        {"words": words[offset : offset + 1000]},
    )

rows = db.execute_and_fetch(
    """UNWIND range(0, size($tokens) - 1) AS position
    MATCH (word:StrideAlignDemoWord)
    WITH position, word.value AS candidate,
         stride_align.needleman_wunsch_normalized_score(
             $tokens[position], word.value
         ) AS score
    ORDER BY position, score DESC, candidate
    WITH position, collect(candidate)[0] AS correction
    RETURN position, correction
    ORDER BY position""",
    {"tokens": tokens},
)

print(" ".join(row["correction"] for row in rows))
